    store_id   SERIAL       PRIMARY KEY,
    store_name VARCHAR(100) NOT NULL,
    city       VARCHAR(50)  NOT NULL,

-- ============================================================
-- UPDATE member_since = first order_date per customer
-- ============================================================
UPDATE customers c
SET member_since = first_order.first_order_date
FROM (
    SELECT customer_id, MIN(order_date) AS first_order_date
    FROM orders
    GROUP BY customer_id
) first_order
WHERE c.customer_id = first_order.customer_id;


-- ============================================================
-- UPDATE ORDER TOTALS
-- ============================================================
UPDATE orders o
SET total_amount = COALESCE(sub.total, 0.00)
FROM (
    SELECT order_id, ROUND(SUM(quantity * unit_price), 2) AS total
    FROM order_items
    GROUP BY order_id
) sub
WHERE o.order_id = sub.order_id;
